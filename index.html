Experiment 3 & 4 · Program 1

Parameterized Carry-Lookahead Adder

N-bit adder (default 16) that computes generate G = a&b and propagate P = a^b, then every carry with c[i+1] = G[i] | P[i]&c[i] inside an always @(*) loop.

PASSTestbench checks: 8 corner cases (all 0s, all 1s, FFFF+1 …) and 10,000 random vectors against the + operator.

cla_adder.v31 lines
// ============================================================
// Program 1 : Parameterized Carry-Lookahead Adder (CLA)
// ------------------------------------------------------------
//  G (generate)  = a & b   -> this bit makes a carry by itself
//  P (propagate) = a ^ b   -> this bit passes an incoming carry
//  Carry equation : c[i+1] = G[i] | (P[i] & c[i])
//  Sum            : s[i]   = P[i] ^ c[i]
//  Change N to get any width (default 16 bits).
// ============================================================
`timescale 1ns/1ps
module cla_adder #(parameter N = 16) (
    input      [N-1:0] a,
    input      [N-1:0] b,
    input              cin,
    output reg [N-1:0] sum,
    output reg         cout
);
    reg [N-1:0] g, p;   // generate and propagate
    reg [N:0]   c;      // carries (c[0] = cin)
    integer i;

    always @(*) begin
        g    = a & b;
        p    = a ^ b;
        c[0] = cin;
        for (i = 0; i < N; i = i + 1)
            c[i+1] = g[i] | (p[i] & c[i]);   // carry-lookahead equation
        sum  = p ^ c[N-1:0];
        cout = c[N];
    end
endmodule
tb_cla_adder.v61 lines
// ============================================================
// Testbench : Carry-Lookahead Adder
//  Reference model : the Verilog '+' operator
//  Tests : directed corner cases + 10,000 random vectors
// ============================================================
`timescale 1ns/1ps
module tb_cla_adder;
    parameter N = 16;

    reg  [N-1:0] a, b;
    reg          cin;
    wire [N-1:0] sum;
    wire         cout;

    reg  [N:0]   expected;     // N+1 bits = {carry, sum}
    integer errors = 0;
    integer tests  = 0;
    integer i;

    cla_adder #(.N(N)) dut (.a(a), .b(b), .cin(cin), .sum(sum), .cout(cout));

    // apply one vector and compare with the '+' operator
    task check;
        input [N-1:0] ta, tb;
        input         tc;
        begin
            a = ta; b = tb; cin = tc;
            #1;
            expected = a + b + cin;
            tests = tests + 1;
            if ({cout, sum} !== expected) begin
                errors = errors + 1;
                $display("ERROR: %h + %h + %b -> got %b_%h, expected %b_%h",
                          a, b, cin, cout, sum, expected[N], expected[N-1:0]);
            end
        end
    endtask

    initial begin
        // ---------- directed corner cases ----------
        check({N{1'b0}}, {N{1'b0}}, 1'b0);   // all zeros
        check({N{1'b0}}, {N{1'b0}}, 1'b1);   // 0 + 0 + 1
        check({N{1'b1}}, {N{1'b1}}, 1'b0);   // all ones
        check({N{1'b1}}, {N{1'b1}}, 1'b1);   // all ones + carry
        check({N{1'b1}}, 1,         1'b0);   // FFFF + 1 : longest carry chain
        check({N{1'b1}}, {N{1'b0}}, 1'b1);   // FFFF + 0 + 1 : carry ripples through
        check({1'b1,{N-1{1'b0}}}, {1'b1,{N-1{1'b0}}}, 1'b0); // 8000 + 8000
        check({N/2{2'b01}}, {N/2{2'b10}}, 1'b1);              // 5555 + AAAA + 1

        // ---------- 10,000 random vectors ----------
        for (i = 0; i < 10000; i = i + 1)
            check($random, $random, $random);

        // ---------- summary ----------
        $display("--------------------------------------");
        $display("CLA (N=%0d): %0d tests, %0d errors", N, tests, errors);
        if (errors == 0) $display("RESULT: PASS");
        else             $display("RESULT: FAIL");
        $finish;
    end
endmodule

16-bit Barrel Shifter

Shifts by 0–15 in five modes: LSL, LSR, ASR (keeps the sign bit), ROL and ROR. Each output bit is picked from the correct input bit, like the barrel shifter's mux network.

PASSTestbench checks: A separate reference function per mode; every mode × every shift amount × 50 data values = 4,000 checks.

barrel_shifter.v36 lines
// ============================================================
// Program 2 : 16-bit Barrel Shifter
// ------------------------------------------------------------
//  mode : 000 = Logical Shift Left   (LSL)
//         001 = Logical Shift Right  (LSR)
//         010 = Arithmetic Shift Right (ASR, keeps sign bit)
//         011 = Rotate Left  (ROL)
//         100 = Rotate Right (ROR)
//         others = pass data through unchanged
//  Each output bit is picked from the right input bit
//  (this is how a barrel shifter's multiplexers work).
// ============================================================
`timescale 1ns/1ps
module barrel_shifter (
    input      [15:0] din,
    input      [3:0]  shamt,   // shift amount 0..15
    input      [2:0]  mode,
    output reg [15:0] dout
);
    localparam LSL = 3'b000, LSR = 3'b001, ASR = 3'b010,
               ROL = 3'b011, ROR = 3'b100;
    integer i;

    always @(*) begin
        for (i = 0; i < 16; i = i + 1) begin
            case (mode)
                LSL: dout[i] = (i >= shamt)      ? din[i - shamt] : 1'b0;
                LSR: dout[i] = (i + shamt <= 15) ? din[i + shamt] : 1'b0;
                ASR: dout[i] = (i + shamt <= 15) ? din[i + shamt] : din[15]; // fill with sign
                ROL: dout[i] = din[(i - shamt + 16) % 16];
                ROR: dout[i] = din[(i + shamt) % 16];
                default: dout[i] = din[i];
            endcase
        end
    end
endmodule

// ============================================================
// Testbench : 16-bit Barrel Shifter
//  One reference function per mode.
//  Every mode x every shift amount (0..15) x 50 random values
// ============================================================
`timescale 1ns/1ps
module tb_barrel_shifter;
    reg  [15:0] din;
    reg  [3:0]  shamt;
    reg  [2:0]  mode;
    wire [15:0] dout;

    reg  [15:0] expected;
    integer m, s, k;
    integer errors = 0, tests = 0;

    barrel_shifter dut (.din(din), .shamt(shamt), .mode(mode), .dout(dout));

    // ---------- reference functions (one per mode) ----------
    function [15:0] ref_lsl(input [15:0] d, input [3:0] n);
        ref_lsl = d << n;
    endfunction

    function [15:0] ref_lsr(input [15:0] d, input [3:0] n);
        ref_lsr = d >> n;
    endfunction

    function [15:0] ref_asr(input [15:0] d, input [3:0] n);
        ref_asr = $signed(d) >>> n;
    endfunction

    function [15:0] ref_rol(input [15:0] d, input [3:0] n);
        ref_rol = (d << n) | (d >> (16 - n));
    endfunction

    function [15:0] ref_ror(input [15:0] d, input [3:0] n);
        ref_ror = (d >> n) | (d << (16 - n));
    endfunction

    function [8*3-1:0] mode_name(input [2:0] md);
        case (md)
            0: mode_name = "LSL";  1: mode_name = "LSR";  2: mode_name = "ASR";
            3: mode_name = "ROL";  4: mode_name = "ROR";  default: mode_name = "???";
        endcase
    endfunction

    initial begin
        for (m = 0; m < 5; m = m + 1)            // every mode
            for (s = 0; s < 16; s = s + 1)       // every shift amount
                for (k = 0; k < 50; k = k + 1) begin   // 50 data values
                    mode  = m;
                    shamt = s;
                    // first two values are fixed corner cases, rest random
                    if (k == 0)      din = 16'h8001;   // sign bit + LSB set
                    else if (k == 1) din = 16'hFFFF;
                    else             din = $random;
                    #1;
                    case (m)
                        0: expected = ref_lsl(din, shamt);
                        1: expected = ref_lsr(din, shamt);
                        2: expected = ref_asr(din, shamt);
                        3: expected = ref_rol(din, shamt);
                        4: expected = ref_ror(din, shamt);
                    endcase
                    tests = tests + 1;
                    if (dout !== expected) begin
                        errors = errors + 1;
                        $display("ERROR %s: din=%h shamt=%0d got=%h expected=%h",
                                 mode_name(mode), din, shamt, dout, expected);
                    end
                end

        $display("--------------------------------------");
        $display("Barrel shifter: %0d tests, %0d errors", tests, errors);
        if (errors == 0) $display("RESULT: PASS");
        else             $display("RESULT: FAIL");
        $finish;
    end
endmodule

8-bit ALU with Status Flags

16 operations selected by a 4-bit opcode (ADD … DEC) with Zero, Carry, signed Overflow and Negative flags. For SUB/SBB/DEC the carry flag means borrow.

PASSTestbench checks: 232 vectors read from alu_vectors.txt with $readmemh; every result and flag compared, mismatches printed with the operation name.

alu8.v79 lines
// ============================================================
// Program 3 : 8-bit ALU with Status Flags
// ------------------------------------------------------------
//  opcode  op     result              carry flag (C)
//   0      ADD    A + B               carry out
//   1      ADC    A + B + cin         carry out
//   2      SUB    A - B               borrow (1 if A < B)
//   3      SBB    A - B - cin         borrow
//   4      AND    A & B               0
//   5      OR     A | B               0
//   6      XOR    A ^ B               0
//   7      NOT    ~A                  0
//   8      NAND   ~(A & B)            0
//   9      NOR    ~(A | B)            0
//   A      SHL    A << 1              bit shifted out (A[7])
//   B      SHR    A >> 1              bit shifted out (A[0])
//   C      ROL    rotate left         A[7]
//   D      ROR    rotate right        A[0]
//   E      INC    A + 1               carry out
//   F      DEC    A - 1               borrow
//  Flags : Z = result is zero
//          C = carry / borrow (see table)
//          V = signed overflow (only for ADD/ADC/SUB/SBB/INC/DEC)
//          N = result is negative (bit 7)
// ============================================================
`timescale 1ns/1ps
module alu8 (
    input      [3:0] opcode,
    input      [7:0] a,
    input      [7:0] b,
    input            cin,
    output reg [7:0] result,
    output           zero,
    output reg       carry,
    output reg       overflow,
    output           negative
);
    reg [8:0] tmp;   // 9 bits: extra bit holds carry / borrow

    always @(*) begin
        tmp      = 9'd0;
        carry    = 1'b0;
        overflow = 1'b0;
        case (opcode)
            4'h0: begin tmp = a + b;                       // ADD
                        result = tmp[7:0]; carry = tmp[8];
                        overflow = (a[7] == b[7]) && (result[7] != a[7]); end
            4'h1: begin tmp = a + b + cin;                 // ADC
                        result = tmp[7:0]; carry = tmp[8];
                        overflow = (a[7] == b[7]) && (result[7] != a[7]); end
            4'h2: begin tmp = {1'b0, a} - {1'b0, b};       // SUB
                        result = tmp[7:0]; carry = tmp[8];  // tmp[8]=1 means borrow
                        overflow = (a[7] != b[7]) && (result[7] != a[7]); end
            4'h3: begin tmp = {1'b0, a} - {1'b0, b} - cin; // SBB
                        result = tmp[7:0]; carry = tmp[8];
                        overflow = (a[7] != b[7]) && (result[7] != a[7]); end
            4'h4: result = a & b;                          // AND
            4'h5: result = a | b;                          // OR
            4'h6: result = a ^ b;                          // XOR
            4'h7: result = ~a;                             // NOT
            4'h8: result = ~(a & b);                       // NAND
            4'h9: result = ~(a | b);                       // NOR
            4'hA: begin result = {a[6:0], 1'b0}; carry = a[7]; end  // SHL
            4'hB: begin result = {1'b0, a[7:1]}; carry = a[0]; end  // SHR
            4'hC: begin result = {a[6:0], a[7]}; carry = a[7]; end  // ROL
            4'hD: begin result = {a[0], a[7:1]}; carry = a[0]; end  // ROR
            4'hE: begin tmp = a + 1;                       // INC
                        result = tmp[7:0]; carry = tmp[8];
                        overflow = (a == 8'h7F); end         // +127 + 1
            4'hF: begin tmp = {1'b0, a} - 1;               // DEC
                        result = tmp[7:0]; carry = tmp[8];
                        overflow = (a == 8'h80); end         // -128 - 1
            default: result = 8'h00;
        endcase
    end

    assign zero     = (result == 8'h00);
    assign negative = result[7];
endmodule

// ============================================================
// Testbench : 8-bit ALU
//  Test vectors are read from "alu_vectors.txt" using $readmemh.
//  Each line = 9 hex digits :  O C AA BB RR F
//     O  = opcode            C  = carry-in (0/1)
//     AA = operand A         BB = operand B
//     RR = expected result   F  = expected flags {Z,C,V,N}
//  Keep alu_vectors.txt in the same folder as the simulation.
// ============================================================
`timescale 1ns/1ps
module tb_alu8;
    reg  [3:0] opcode;
    reg  [7:0] a, b;
    reg        cin;
    wire [7:0] result;
    wire       zero, carry, overflow, negative;

    reg  [35:0] vectors [0:511];   // room for up to 512 vectors
    reg  [7:0]  exp_result;
    reg  [3:0]  exp_flags;
    wire [3:0]  flags = {zero, carry, overflow, negative};
    integer i, errors, tests;

    alu8 dut (.opcode(opcode), .a(a), .b(b), .cin(cin), .result(result),
              .zero(zero), .carry(carry), .overflow(overflow), .negative(negative));

    // name of each operation (used in error messages)
    function [8*4-1:0] op_name(input [3:0] op);
        case (op)
            4'h0: op_name = "ADD";  4'h1: op_name = "ADC";  4'h2: op_name = "SUB";
            4'h3: op_name = "SBB";  4'h4: op_name = "AND";  4'h5: op_name = "OR";
            4'h6: op_name = "XOR";  4'h7: op_name = "NOT";  4'h8: op_name = "NAND";
            4'h9: op_name = "NOR";  4'hA: op_name = "SHL";  4'hB: op_name = "SHR";
            4'hC: op_name = "ROL";  4'hD: op_name = "ROR";  4'hE: op_name = "INC";
            default: op_name = "DEC";
        endcase
    endfunction

    initial begin
        // (a "not enough words" warning here is normal: file is smaller than 512)
        $readmemh("alu_vectors.txt", vectors);
        errors = 0;
        tests  = 0;
        i      = 0;

        // run until we reach an empty (unloaded = xxx) entry
        while (i < 512 && vectors[i] !== 36'bx) begin
            {opcode, cin, a, b, exp_result, exp_flags} =
                {vectors[i][35:32], vectors[i][28], vectors[i][27:20],
                 vectors[i][19:12], vectors[i][11:4], vectors[i][3:0]};
            #1;
            tests = tests + 1;
            if (result !== exp_result || flags !== exp_flags) begin
                errors = errors + 1;
                $display("MISMATCH %-4s A=%h B=%h cin=%b : got R=%h ZCVN=%b, expected R=%h ZCVN=%b",
                         op_name(opcode), a, b, cin, result, flags, exp_result, exp_flags);
            end
            i = i + 1;
        end

        $display("--------------------------------------");
        $display("ALU: %0d vectors checked, %0d mismatches", tests, errors);
        if (tests == 0)        $display("RESULT: FAIL (alu_vectors.txt not found or empty)");
        else if (errors == 0)  $display("RESULT: PASS");
        else                   $display("RESULT: FAIL");
        $finish;
    end

Signed Booth Multiplier (Radix-2)

8×8 signed multiplier using Booth's rule on {Q0, Q-1}: 01 → add M, 10 → subtract M, then arithmetic shift right, 8 times. The accumulator is 9 bits so −128 × −128 works.

PASSTestbench checks: Exhaustive: all 65,536 signed pairs against the * operator, plus −128×−128, −128×127 and ×0 printed.

booth_mult.v41 lines
// ============================================================
// Program 4 : 8x8 Signed Booth Multiplier (Radix-2, combinational)
// ------------------------------------------------------------
//  Booth rule : look at {Q[0], Q_-1}
//     01 -> A = A + M        10 -> A = A - M
//     00 / 11 -> do nothing
//  then arithmetic-shift-right {A, Q, Q_-1} by one bit.
//  Repeat 8 times (once per multiplier bit).
//
//  A is 9 bits (not 8) so that  -128 x -128 does not overflow
//  while adding/subtracting M (M = -128 needs 9 bits for -M).
// ============================================================
`timescale 1ns/1ps

Hamming (7,4) Error-Correcting Codec

Encoder adds parity bits p1, p2, p3 to 4 data bits. Decoder computes the syndrome {s3,s2,s1}, which is the position of a wrong bit, and flips it back.

hamming74.v56 lines
// ============================================================
// Program 5 : Hamming (7,4) Encoder and Decoder
// ------------------------------------------------------------
//  Codeword bit positions 1..7  =  p1 p2 d1 p3 d2 d3 d4
//  (stored as code[0] = position 1 ... code[6] = position 7)
//     p1 = d1 ^ d2 ^ d4      (checks positions 1,3,5,7)
//     p2 = d1 ^ d3 ^ d4      (checks positions 2,3,6,7)
//     p3 = d2 ^ d3 ^ d4      (checks positions 4,5,6,7)
//  Decoder : syndrome {s3,s2,s1} = position of the wrong bit
//            (0 means no error). Flip that bit to correct it.
// ============================================================
`timescale 1ns/1ps

// ---------------- Encoder ----------------
module hamming74_encoder (
    input      [3:0] data,      // data = {d4, d3, d2, d1}
    output reg [6:0] code
);
    reg d1, d2, d3, d4;
    always @(*) begin
        {d4, d3, d2, d1} = data;
        code[0] = d1 ^ d2 ^ d4;   // position 1 : p1
        code[1] = d1 ^ d3 ^ d4;   // position 2 : p2
        code[2] = d1;             // position 3
        code[3] = d2 ^ d3 ^ d4;   // position 4 : p3
        code[4] = d2;             // position 5
        code[5] = d3;             // position 6
        code[6] = d4;             // position 7
    end
endmodule

// ---------------- Decoder ----------------
module hamming74_decoder (
    input      [6:0] code_in,   // received codeword (may have 1 error)
    output reg [3:0] data_out,  // corrected data
    output reg [2:0] err_pos,   // 0 = no error, 1..7 = bit position fixed
    output           error      // 1 if an error was corrected
);
    reg [6:0] fixed;
    reg s1, s2, s3;

    always @(*) begin
        s1 = code_in[0] ^ code_in[2] ^ code_in[4] ^ code_in[6]; // pos 1,3,5,7
        s2 = code_in[1] ^ code_in[2] ^ code_in[5] ^ code_in[6]; // pos 2,3,6,7
        s3 = code_in[3] ^ code_in[4] ^ code_in[5] ^ code_in[6]; // pos 4,5,6,7
        err_pos = {s3, s2, s1};

        fixed = code_in;
        if (err_pos != 0)
            fixed[err_pos - 1] = ~fixed[err_pos - 1];   // flip the bad bit

        data_out = {fixed[6], fixed[5], fixed[4], fixed[2]};  // {d4,d3,d2,d1}
    end

    assign error = (err_pos != 3'd0);
endmodule

divider8.v45 lines
// ============================================================
// Program 6 : 8-bit Unsigned Combinational Divider (Restoring)
// ------------------------------------------------------------
//  Restoring division, repeated 8 times:
//    1. shift {R, Q} left by one bit
//    2. R = R - divisor
//    3. if R is negative -> restore R (add divisor back), Q[0] = 0
//       else                                             Q[0] = 1
//  Divide by zero : div_by_zero = 1,
//                   quotient = 8'hFF, remainder = dividend
// ============================================================
`timescale 1ns/1ps
module divider8 (
    input      [7:0] dividend,
    input      [7:0] divisor,
    output reg [7:0] quotient,
    output reg [7:0] remainder,
    output reg       div_by_zero
);
    reg [8:0] R;     // partial remainder (9 bits : MSB is the sign)
    reg [7:0] Q;
    integer i;

    always @(*) begin
        if (divisor == 8'd0) begin
            div_by_zero = 1'b1;
            quotient    = 8'hFF;       // defined value for divide by zero
            remainder   = dividend;
        end else begin
            div_by_zero = 1'b0;
            R = 9'd0;
            Q = dividend;
            for (i = 0; i < 8; i = i + 1) begin
                {R, Q} = {R[7:0], Q, 1'b0};        // step 1 : shift left
                R = R - {1'b0, divisor};           // step 2 : subtract
                if (R[8] == 1'b1)                  // step 3 : negative?
                    R = R + {1'b0, divisor};       //   restore, Q[0] stays 0
                else
                    Q[0] = 1'b1;                   //   keep, Q[0] = 1
            end
            quotient  = Q;
            remainder = R[7:0];
        end
    end
endmodule

// ============================================================
// Program 7 : Two-Digit BCD Adder / Subtractor
// ------------------------------------------------------------
//  a, b : packed BCD  {tens digit, ones digit}  e.g. 8'h47 = 47
//  sub = 0 : result = a + b + cin        cout = carry out  (1 if > 99)
//  sub = 1 : result = a - b - cin        cout = borrow out (1 if < 0)
//            (cin works as borrow-in; when borrow = 1 the result
//             is the 10's complement, e.g. 03 - 05 = 98, borrow 1)
//  Subtraction : a - b = a + (9's complement of b) + 1
//  Decimal correction : if a digit sum > 9, add 6
//  invalid = 1 if any input digit is greater than 9 (result = 00)
// ============================================================
`timescale 1ns/1ps
module bcd_addsub (
    input      [7:0] a,
    input      [7:0] b,
    input            cin,       // carry-in (add) / borrow-in (sub)
    input            sub,       // 0 = add, 1 = subtract
    output reg [7:0] result,
    output reg       cout,      // carry-out (add) / borrow-out (sub)
    output           invalid
);
    reg [3:0] b_lo, b_hi;       // b, or 9's complement of b
    reg [4:0] lo, hi;           // {carry, digit}

    // add two BCD digits + carry, with decimal correction
    function [4:0] digit_add;
        input [3:0] x, y;
        input       c;
        reg   [4:0] s;
        begin
            s = x + y + c;
            if (s > 9)
                s = s + 6;          // decimal correction
            digit_add = s;          // s[4] = carry, s[3:0] = BCD digit
        end
    endfunction

    // any digit above 9 is not BCD
    assign invalid = (a[3:0] > 9) || (a[7:4] > 9) || (b[3:0] > 9) || (b[7:4] > 9);

    always @(*) begin
        if (sub) begin
            b_lo = 4'd9 - b[3:0];        // 9's complement of each digit
            b_hi = 4'd9 - b[7:4];
            lo   = digit_add(a[3:0], b_lo, ~cin);   // +1 (unless borrow-in)
        end else begin
            b_lo = b[3:0];
            b_hi = b[7:4];
            lo   = digit_add(a[3:0], b_lo, cin);
        end
        hi = digit_add(a[7:4], b_hi, lo[4]);

        result = {hi[3:0], lo[3:0]};
        cout   = sub ? ~hi[4] : hi[4];   // no final carry in subtraction = borrow

        if (invalid) begin
            result = 8'h00;
            cout   = 1'b0;
        end
    end

// ============================================================
// Program 2 : Parameterized Universal Shift Register (N bits)
// ------------------------------------------------------------
//  ctrl   asr   operation
//   00     -    hold
//   01     0    shift right  (sin_r enters at the MSB)
//   01     1    ARITHMETIC shift right (MSB copied, keeps sign)
//   10     -    shift left   (sin_l enters at the LSB)
//   11     -    parallel load (q = d)
//  rst_n : asynchronous reset (active low)
//  clr   : synchronous clear  (q = 0 on next clock)
//  Priority : rst_n > clr > ctrl
// ============================================================

module universal_shift_reg #(parameter N = 8) (
    input              clk,
    input              rst_n,
    input              clr,
    input      [1:0]   ctrl,

    input              sin_r,     // serial input for shift right
    input              sin_l,     // serial input for shift left
    input      [N-1:0] d,         // parallel data
    output reg [N-1:0] q
);

        if (!rst_n)
            q <= {N{1'b0}};
        else if (clr)
            q <= {N{1'b0}};
        else begin
            case (ctrl)
                2'b00: q <= q;                                  // hold

                2'b10: q <= {q[N-2:0], sin_l};                  // shift left
                2'b11: q <= d;                                  // parallel load
            endcase
        end
    end
endmodule

modn_counter.v52 lines
// ============================================================
// Program 4 : Programmable Modulo-N Up/Down Counter (8-bit)
// ------------------------------------------------------------
//  Counts 0 .. mod_val-1 (mod_val = 2..255, any value).
//  Priority : rst_n (async) > load > en
//  up = 1 : ... mod_val-2, mod_val-1, 0, 1 ...   (tc = 1 on wrap)
//  up = 0 : ... 1, 0, mod_val-1, mod_val-2 ...   (tc = 1 on wrap)
//  tc is high for exactly one clock cycle after each wraparound.
//
//  Rules for special cases (our design choice):
//   1. load_val >= mod_val  ->  counter is loaded with 0
//   2. mod_val changed so that count >= mod_val  -> on the next
//      enabled clock the counter goes to 0 (up) or mod_val-1
//      (down). This is a recovery, not a wrap, so tc stays 0.
//   3. mod_val of 0 or 1 is treated as 2.
// ============================================================
`timescale 1ns/1ps
module modn_counter (
    input            clk,
    input            rst_n,      // async reset
    input            en,         // count enable
    input            up,         // 1 = up, 0 = down
    input            load,       // synchronous load
    input      [7:0] load_val,
    input      [7:0] mod_val,
    output reg [7:0] count,
    output reg       tc          // terminal count pulse
);
    wire [7:0] m = (mod_val < 2) ? 8'd2 : mod_val;   // effective modulus

    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            count <= 8'd0;
            tc    <= 1'b0;
        end else if (load) begin
            count <= (load_val < m) ? load_val : 8'd0;   // rule 1
            tc    <= 1'b0;
        end else if (en) begin
            if (count >= m) begin                        // rule 2
                count <= up ? 8'd0 : m - 1;
                tc    <= 1'b0;
            end else if (up) begin
                if (count == m - 1) begin count <= 8'd0;      tc <= 1'b1; end
                else                begin count <= count + 1; tc <= 1'b0; end
            end else begin
                if (count == 0)     begin count <= m - 1;     tc <= 1'b1; end
                else                begin count <= count - 1; tc <= 1'b0; end
            end
        end else
            tc <= 1'b0;
    end
endmodule

johnson_counter.v46 lines
// ============================================================
// Program 5 : Self-Correcting 8-bit Johnson (Twisted-Ring) Counter
// ------------------------------------------------------------
//  Normal step : q <= {q[6:0], ~q[7]}   (shift left, inverted MSB in)
//  Valid sequence (16 states):
//    00000000 -> 00000001 -> 00000011 -> ... -> 11111111
//    -> 11111110 -> 11111100 -> ... -> 10000000 -> 00000000
//  A valid state has AT MOST ONE place where neighbouring bits
//  differ (q[i] != q[i+1]).  The other 240 states are illegal.
//  Correction : if the state is illegal, go to 00000000.
//               => back in the valid sequence within 1 clock.
//  load / load_val is used by the testbench to start anywhere.
// ============================================================
`timescale 1ns/1ps
module johnson_counter (
    input            clk,
    input            rst_n,
    input            load,
    input      [7:0] load_val,
    output reg [7:0] q,
    output           valid        // 1 when q is one of the 16 legal states
);
    // count how many neighbouring bit pairs are different
    function is_valid(input [7:0] s);
        integer i, changes;
        begin
            changes = 0;
            for (i = 0; i < 7; i = i + 1)
                if (s[i] != s[i+1]) changes = changes + 1;
            is_valid = (changes <= 1);
        end
    endfunction

    assign valid = is_valid(q);

    always @(posedge clk or negedge rst_n) begin
        if (!rst_n)
            q <= 8'd0;
        else if (load)
            q <= load_val;
        else if (!valid)
            q <= 8'd0;                    // self-correction
        else
            q <= {q[6:0], ~q[7]};         // normal Johnson step
    end
endmodule

// ------------------------------------------------------------
//  Polynomial : x^16 + x^14 + x^13 + x^11 + 1
//  Parameter TYPE selects the form:
//    TYPE = 0 : Fibonacci (external XOR)
//               new bit = q[0]^q[2]^q[3]^q[5], shifted in at the MSB

    input             load,       // load the seed
    input      [15:0] seed,

        else

// ------------------------------------------------------------
//  - counts in binary inside (bin)

//   1. exactly one output bit changes per increment
//      (and none when en = 0)

Programmable 50% Duty-Cycle Clock Divider

Divides by any N from 1 to 255 at runtime. Odd N combines a rising-edge and a falling-edge register to get exactly 50%. A new N only takes effect at the end of a full period.

PASSTestbench checks: Period and high time measured with $realtime for every N = 1…255, plus a glitch monitor during 200 random changes of N.

clock_divider.v67 lines
// ============================================================
// Program 9 : Programmable Clock Divider, 50% duty cycle
// ------------------------------------------------------------
//  clk_out frequency = clk_in / N      (N = 1..255, at runtime)
//
//  How it works
//   cnt counts 0 .. N-1 on the RISING edge of clk_in.
//   pos_q is high while cnt < N/2 (integer divide).
//   EVEN N : clk_out = pos_q              high = N/2 cycles
//   ODD  N : neg_q = pos_q delayed by half a cycle (falling edge)
//            clk_out = pos_q | neg_q      high = (N-1)/2 + 0.5 = N/2
//   N = 1  : clk_out = clk_in (no division, pos_q = neg_q = 0)
//
//  Glitch-free change :
//   - a new N is only taken (n_act <= n_in) at the end of a
//     complete output period, so no short (runt) pulse is made.
//   - the "use clk_in" select (sel1) changes on the FALLING edge,
//     and the paths are combined with OR, so the output never
//     drops while switching into or out of N = 1.
//  N = 0 is treated as N = 1.
// ============================================================
`timescale 1ns/1ps
module clock_divider (
    input        clk_in,
    input        rst_n,
    input  [7:0] n_in,        // requested divide value
    output       clk_out
);
    reg [7:0] n_act;          // divide value in use now
    reg [7:0] cnt;
    reg       pos_q, neg_q;
    reg       sel1;           // 1 = pass clk_in straight through (N = 1)

    wire [7:0] n_new   = (n_in == 0) ? 8'd1 : n_in;
    wire       end_per = (cnt >= n_act - 1);        // last cycle of the period

    // ---------- rising-edge counter ----------
    always @(posedge clk_in or negedge rst_n) begin
        if (!rst_n) begin
            cnt   <= 0;
            n_act <= 1;
            pos_q <= 0;
        end else if (end_per) begin
            cnt   <= 0;
            n_act <= n_new;                         // take new N here only
            pos_q <= (n_new >= 2);                  // new period starts high
        end else begin
            cnt   <= cnt + 1;
            pos_q <= ((cnt + 1) < (n_act >> 1));    // high for first N/2 cycles
        end
    end

    // ---------- falling-edge registers ----------
    always @(negedge clk_in or negedge rst_n) begin
        if (!rst_n) begin
            neg_q <= 0;
            sel1  <= 1;
        end else begin
            neg_q <= pos_q;               // half-cycle delayed copy (odd N)
            sel1  <= (n_act == 1);        // N = 1 select, changed while clk_in is low
        end
    end

    assign clk_out = (sel1 & clk_in)            // N = 1
                   | pos_q                      // even and odd N
                   | (n_act[0] & neg_q);        // extra half cycle for odd N
endmodule

Synchronous FIFO with Status Flags

16 × 8 FIFO using pointers with an extra wrap bit. Full, empty, almost-full, almost-empty and count, plus overflow/underflow flags when a write or read is refused.

sync_fifo.v71 lines
// ============================================================
// Program 10 : Synchronous FIFO with Status Flags
// ------------------------------------------------------------
//  DEPTH x WIDTH memory (default 16 x 8)
//  Pointers have ONE EXTRA bit (the "wrap bit"):
//    empty : wr_ptr == rd_ptr               (same address, same wrap)
//    full  : addresses equal, wrap bits different
//    count = wr_ptr - rd_ptr                (number of words stored)
//  almost_full  : count >= AF_LEVEL
//  almost_empty : count <= AE_LEVEL
//  Write when full  -> ignored, overflow  = 1 for one cycle
//  Read  when empty -> ignored, underflow = 1 for one cycle
//  rd_data is updated on the clock edge of a successful read.
// ============================================================
`timescale 1ns/1ps
module sync_fifo #(
    parameter DEPTH    = 16,
    parameter WIDTH    = 8,
    parameter AF_LEVEL = 14,      // almost-full threshold
    parameter AE_LEVEL = 2        // almost-empty threshold
) (
    input                    clk,
    input                    rst_n,
    input                    wr_en,
    input      [WIDTH-1:0]   wr_data,
    input                    rd_en,
    output reg [WIDTH-1:0]   rd_data,
    output                   full,
    output                   empty,
    output                   almost_full,
    output                   almost_empty,
    output     [$clog2(DEPTH):0] count,
    output reg               overflow,
    output reg               underflow
);
    localparam AW = $clog2(DEPTH);        // address width (4 for 16)

    reg [WIDTH-1:0] mem [0:DEPTH-1];
    reg [AW:0]      wr_ptr, rd_ptr;       // AW+1 bits : extra wrap bit

    assign empty        = (wr_ptr == rd_ptr);
    assign full         = (wr_ptr[AW] != rd_ptr[AW]) &&
                          (wr_ptr[AW-1:0] == rd_ptr[AW-1:0]);
    assign count        = wr_ptr - rd_ptr;
    assign almost_full  = (count >= AF_LEVEL);
    assign almost_empty = (count <= AE_LEVEL);

    wire do_write = wr_en && !full;
    wire do_read  = rd_en && !empty;

    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            wr_ptr    <= 0;
            rd_ptr    <= 0;
            rd_data   <= 0;
            overflow  <= 0;
            underflow <= 0;
        end else begin
            if (do_write) begin
                mem[wr_ptr[AW-1:0]] <= wr_data;
                wr_ptr <= wr_ptr + 1;
            end
            if (do_read) begin
                rd_data <= mem[rd_ptr[AW-1:0]];
                rd_ptr  <= rd_ptr + 1;
            end
            overflow  <= wr_en && full;       // error flags
            underflow <= rd_en && empty;
        end
    end
endmodule

// ============================================================
// Testbench : Synchronous FIFO with Scoreboard
//  - scoreboard = a plain Verilog array used as a queue
//    (head = next to read, tail = next to write, sb_count)
//  - 20,000 cycles of random simultaneous reads/writes
//  - write/read probability changes every 1,000 cycles
//    (write-heavy -> fills up, read-heavy -> drains)
//  - every cycle checks : read data order, full, empty,
//    almost_full, almost_empty, count, overflow, underflow
// ============================================================
`timescale 1ns/1ps
module tb_sync_fifo;
    parameter DEPTH = 16, WIDTH = 8, AF = 14, AE = 2;

    reg              clk = 0, rst_n = 0, wr_en = 0, rd_en = 0;
    reg  [WIDTH-1:0] wr_data = 0;
    wire [WIDTH-1:0] rd_data;
    wire             full, empty, almost_full, almost_empty, overflow, underflow;
    wire [4:0]       count;

    // ---------- scoreboard ----------
    reg  [WIDTH-1:0] sb [0:DEPTH-1];
    integer head = 0, tail = 0, sb_count = 0;
    reg  [WIDTH-1:0] exp_data = 0;
    reg              exp_ovf = 0, exp_unf = 0, do_wr, do_rd;

    integer cyc, wr_pct, rd_pct, errors = 0;
    integer n_full = 0, n_empty = 0, n_ovf = 0, n_unf = 0, n_reads = 0;

    sync_fifo #(.DEPTH(DEPTH), .WIDTH(WIDTH), .AF_LEVEL(AF), .AE_LEVEL(AE)) dut (
        .clk(clk), .rst_n(rst_n), .wr_en(wr_en), .wr_data(wr_data), .rd_en(rd_en),
        .rd_data(rd_data), .full(full), .empty(empty), .almost_full(almost_full),
        .almost_empty(almost_empty), .count(count), .overflow(overflow), .underflow(underflow));

    always #5 clk = ~clk;

    // ---------- scoreboard update on each clock edge ----------
    always @(posedge clk) begin
        if (rst_n) begin
            exp_ovf = wr_en && (sb_count == DEPTH);   // write while full
            exp_unf = rd_en && (sb_count == 0);       // read while empty
            do_rd   = rd_en && (sb_count != 0);
            do_wr   = wr_en && (sb_count != DEPTH);
            if (do_rd) begin                          // pop from the front
                exp_data = sb[head];
                head     = (head + 1) % DEPTH;
                n_reads  = n_reads + 1;
            end
            if (do_wr) begin                          // push at the back
                sb[tail] = wr_data;
                tail     = (tail + 1) % DEPTH;
            end
            sb_count = sb_count + do_wr - do_rd;
        end
    end

    // ---------- compare DUT with scoreboard ----------
    task check;
        begin
            if (rd_data      !== exp_data            ||
                count        !== sb_count            ||
                full         !== (sb_count == DEPTH) ||
                empty        !== (sb_count == 0)     ||
                almost_full  !== (sb_count >= AF)    ||
                almost_empty !== (sb_count <= AE)    ||
                overflow     !== exp_ovf             ||
                underflow    !== exp_unf) begin
                errors = errors + 1;
                if (errors <= 10)
                    $display("ERROR at %0t: rd_data=%h (exp %h) count=%0d (exp %0d) F=%b E=%b AF=%b AE=%b OVF=%b UNF=%b",
                             $time, rd_data, exp_data, count, sb_count,
                             full, empty, almost_full, almost_empty, overflow, underflow);
            end
            if (full)      n_full  = n_full + 1;
            if (empty)     n_empty = n_empty + 1;
            if (overflow)  n_ovf   = n_ovf + 1;
            if (underflow) n_unf   = n_unf + 1;
        end
    endtask

    initial begin
        #12 rst_n = 1;

        for (cyc = 0; cyc < 20000; cyc = cyc + 1) begin
            @(negedge clk);
            check;                               // result of the last edge
            // change the traffic pattern every 1000 cycles
            case ((cyc / 1000) % 4)
                0: begin wr_pct = 80; rd_pct = 20; end   // fill up -> full
                1: begin wr_pct = 50; rd_pct = 50; end   // balanced
                2: begin wr_pct = 20; rd_pct = 80; end   // drain   -> empty
                3: begin wr_pct = 60; rd_pct = 60; end   // busy
            endcase
            wr_en   = ({$random} % 100) < wr_pct;
            rd_en   = ({$random} % 100) < rd_pct;
            wr_data = $random;
        end
        @(negedge clk) check;

        $display("--------------------------------------");
        $display("Cycles: 20000   successful reads checked: %0d", n_reads);
        $display("Cycles full: %0d  empty: %0d  overflows: %0d  underflows: %0d",
                 n_full, n_empty, n_ovf, n_unf);
        $display("Errors: %0d", errors);
        if (errors == 0 && n_full > 0 && n_empty > 0) $display("RESULT: PASS");
        else                                          $display("RESULT: FAIL");
        $finish;
    end
endmodule
